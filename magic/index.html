const initWebGL = () => {
const engine = window.__MonolithEngine;
const scene = new THREE.Scene();
scene.fog = new THREE.FogExp2(0x010102, 0.08);
engine.scene = scene;

const camera = new THREE.PerspectiveCamera(60, window.innerWidth / window.innerHeight, 0.1, 1000);
camera.position.z = 5;
engine.camera = camera;

const canvas = document.getElementById('xrCanvas');
const renderer = new THREE.WebGLRenderer({ canvas: canvas, antialias: true, alpha: true });
renderer.setSize(window.innerWidth, window.innerHeight);

// --- CENTRAL NEURAL KNOT (Polymorph Base) ---
const material = new THREE.MeshStandardMaterial({
color: 0x007bff, roughness: 0.1, metalness: 0.9, wireframe: true, emissive: 0x001122
});
engine.material = material;

// High vertex count to allow smooth, complex mutation
const geometry = new THREE.TorusKnotGeometry(1.2, 0.35, 256, 64);

// Store original positions for calculating delta without cumulative distortion
const positions = geometry.attributes.position.array;
geometry.userData.originalPositions = new Float32Array(positions.length);
for(let i=0; i<positions.length; i++) {
geometry.userData.originalPositions[i] = positions[i];
}

const knot = new THREE.Mesh(geometry, material);
scene.add(knot);
engine.knot = knot;

// --- DATA STREAMS (Websocket) ---
const cCount = engine.clusterCount;
const nodePositions = new Float32Array(cCount * 3);
const nodeColors = new Float32Array(cCount * 3);
const vels = [];
const isDataActive = new Array(cCount).fill(false);

for(let i=0; i < cCount; i++) {
nodePositions[i*3] = (Math.random() - 0.5) * 30;
nodePositions[i*3+1] = (Math.random() - 0.5) * 30;
nodePositions[i*3+2] = (Math.random() - 0.5) * 15 - 5;
vels.push({ x: (Math.random() - 0.5) * 0.05, y: (Math.random() - 0.5) * 0.05, z: (Math.random() - 0.5) * 0.05 });
}

const nodeGeom = new THREE.BufferGeometry();
nodeGeom.setAttribute('position', new THREE.BufferAttribute(nodePositions, 3));
nodeGeom.setAttribute('color', new THREE.BufferAttribute(nodeColors, 3));
const nodeMat = new THREE.PointsMaterial({ size: 0.15, vertexColors: true, transparent: true, blending: THREE.AdditiveBlending });
engine.clusterNodes = new THREE.Points(nodeGeom, nodeMat);
scene.add(engine.clusterNodes);

const dirLight = new THREE.DirectionalLight(0x00ffff, 2);
dirLight.position.set(5, 5, 5);
scene.add(dirLight);

// Traffic Generator
setInterval(() => {
if(!engine.isAbsorbvoidActive) {
const streams = ['wss://sys.omi', 'mhtml_source_parse', 'neural_node_ping'];
let spawned = 0;
for(let i=0; i<cCount; i++) {
if(!isDataActive[i] && Math.random() > 0.96) {
isDataActive[i] = true;
nodeColors[i*3] = 0; nodeColors[i*3+1] = 1; nodeColors[i*3+2] = 0.8;
nodePositions[i*3] = (Math.random() - 0.5) * 30;
nodePositions[i*3+1] = (Math.random() - 0.5) * 30;
spawned++;
if(spawned > 5) break;
}
}
if(spawned > 0) {
engine.clusterNodes.geometry.attributes.color.needsUpdate = true;
engine.clusterNodes.geometry.attributes.position.needsUpdate = true;
engine.activeDataNodes += spawned;
document.getElementById('metric-nodes').innerText = engine.activeDataNodes;

const ul = document.getElementById('wss-stream-list');
ul.innerHTML = `<li>[${new Date().toLocaleTimeString()}] INCOMING: ${streams[Math.floor(Math.random()*streams.length)]}</li>` + ul.innerHTML;

}
}
}, 600);

// --- POLYMORPH RENDER LOOP ---
function render() {
const time = engine.clock.getElapsedTime();

// Smoothly interpolate mutation level
engine.currentMutation = THREE.MathUtils.lerp(engine.currentMutation, engine.targetMutation, 0.02);
document.getElementById('metric-mutation').innerText = (engine.currentMutation * 100).toFixed(1) + "%";

if(engine.knot) {
// Base Rotation
engine.knot.rotation.x = time * 0.15;
engine.knot.rotation.y = time * 0.2;

// --- POLYMORPH ALGORITHM ---
// Mutates the vertices to transform from smooth knot to chaotic neural web
if(engine.currentMutation > 0.001) {
const posAttr = engine.knot.geometry.attributes.position;
const orig = engine.knot.geometry.userData.originalPositions;
const mutation = engine.currentMutation;

for(let i=0; i < posAttr.count; i++) {
let x = orig[i*3], y = orig[i*3+1], z = orig[i*3+2];

// Procedural geometric noise calculation
let noiseX = Math.sin(y * 4.0 + time * 2.0) * Math.cos(z * 4.0 + time);
let noiseY = Math.cos(x * 4.0 + time * 2.0) * Math.sin(z * 4.0 + time);
let noiseZ = Math.sin(x * 4.0 + time) * Math.cos(y * 4.0 + time * 2.0);

// Calculate spike projection based on vertex distance from center
let dist = Math.sqrt(x*x + y*y + z*z);
let spikeFactor = Math.abs(Math.sin(dist * 10.0 - time * 3.0));

posAttr.setXYZ(i,
x + (noiseX * 0.8 * mutation) + (x * spikeFactor * 0.3 * mutation),
y + (noiseY * 0.8 * mutation) + (y * spikeFactor * 0.3 * mutation),
z + (noiseZ * 0.8 * mutation) + (z * spikeFactor * 0.3 * mutation)
);
}
posAttr.needsUpdate = true;
}

// Dynamic Color Shift based on Mutation (Blue -> Deep Void Purple/Red)
const baseColor = new THREE.Color(0x007bff);
const mutColor = new THREE.Color(0xff00aa); // Vibrant mutation color
engine.material.color.lerpColors(baseColor, mutColor, engine.currentMutation);
engine.material.emissive.lerpColors(new THREE.Color(0x001122), new THREE.Color(0x4a0080), engine.currentMutation);
}

// Particle Assimilation physics
const positions = engine.clusterNodes.geometry.attributes.position.array;
const colors = engine.clusterNodes.geometry.attributes.color.array;
const absorb = engine.isAbsorbvoidActive;

for(let i=0; i < cCount; i++) {
if(!isDataActive[i]) continue;

let px = positions[i*3], py = positions[i*3+1], pz = positions[i*3+2];
if(absorb) {
positions[i*3] += (0 - px) * 0.06;
positions[i*3+1] += (0 - py) * 0.06;
positions[i*3+2] += (0 - pz) * 0.06;

// On Absorption Impact
if(Math.abs(px) < 1.5 && Math.abs(py) < 1.5 && Math.abs(pz) < 1.5) {
isDataActive[i] = false;
colors[i*3] = 0; colors[i*3+1] = 0; colors[i*3+2] = 0;
engine.activeDataNodes--;
engine.neuralLevel++;

// Increase Target Mutation as data is absorbed (caps around 1.5 multiplier)
engine.targetMutation = Math.min(1.5, engine.neuralLevel * 0.05);

document.getElementById('metric-nodes').innerText = engine.activeDataNodes;
document.getElementById('metric-lvl').innerText = engine.neuralLevel;
}
} else {
positions[i*3] += vels[i].x; positions[i*3+1] += vels[i].y; positions[i*3+2] += vels[i].z;
}
}
engine.clusterNodes.geometry.attributes.position.needsUpdate = true;
engine.clusterNodes.geometry.attributes.color.needsUpdate = true;

renderer.render(scene, camera);
requestAnimationFrame(render);
}
render();

window.addEventListener('resize', () => {
camera.aspect = window.innerWidth / window.innerHeight; camera.updateProjectionMatrix();
renderer.setSize(window.innerWidth, window.innerHeight);
});
